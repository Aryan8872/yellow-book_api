import { IsEnum, IsOptional, IsDateString, IsDateString, validateSync } from 'class-validator';
import { Type } from 'class-transformer';

export enum TimeRange {
  LAST_7_DAYS = '7d',
  LAST_30_DAYS = '30d',
  CUSTOM = 'custom',
}

export class TimeRangeDto {
  @IsEnum(TimeRange)
  @Type(() => String)
  range: TimeRange = TimeRange.LAST_7_DAYS;

  @IsOptional()
  @IsDateString()
  startDate?: string;

  @IsOptional()
  @IsDateString()
  endDate?: string;
}

### Task 3: Create Analytics Service

**Type:** scaffold
**Files:** `src/modules/analytics/analytics.service.ts`

**Action:**
Create analytics service with time range helper:
```typescript
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../infrastructure/prisma/prisma.service';
import { TimeRangeDto, TimeRange } from './dto/time-range.dto';

@Injectable()
export class AnalyticsService {
  constructor(private prisma: PrismaService) {}

  getTimeRange(dto: TimeRangeDto): { startDate: Date; endDate: Date } {
    const now = new Date();
    const endDate = new Date(now);
    endDate.setHours(23, 59, 59, 999);

    let startDate: Date;

    if (dto.range === TimeRange.LAST_7_DAYS) {
      startDate = new Date(now);
      startDate.setDate(startDate.getDate() - 7);
      startDate.setHours(0, 0, 0, 0);
    } else if (dto.range === TimeRange.LAST_30_DAYS) {
      startDate = new Date(now);
      startDate.setDate(startDate.getDate() - 30);
      startDate.setHours(0, 0, 0, 0);
    } else if (dto.range === TimeRange.CUSTOM) {
      if (!dto.startDate || !dto.endDate) {
        throw new Error('Custom range requires startDate and endDate');
      }
      startDate = new Date(dto.startDate);
      const customEndDate = new Date(dto.endDate);
      if (startDate > customEndDate) {
        throw new Error('startDate must be before endDate');
      }
      return { startDate, endDate: customEndDate };
    }

    return { startDate, endDate };
  }
}
```

**Verify:**
- Service created with PrismaService injection
- Time range helper method implemented
- Validation for custom ranges
- Default 7d and 30d ranges work

**Done:** Analytics service created
