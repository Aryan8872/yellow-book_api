### Task 2: Implement List Users

Add listUsers method to admin service:
```typescript
async listUsers(dto: ListUsersDto) {
  const where: any = {};

  if (dto.role) {
    where.role = dto.role;
  }

  if (dto.isActive !== undefined) {
    where.isActive = dto.isActive;
  }

  if (dto.search) {
    where.OR = [
      { email: { contains: dto.search, mode: 'insensitive' } },
      { name: { contains: dto.search, mode: 'insensitive' } },
    ];
  }

  const skip = ((dto.page || 1) - 1) * (dto.limit || 20);

  const [users, total] = await Promise.all([
    this.prisma.user.findMany({
      where,
      skip,
      take: dto.limit || 20,
      select: {
        id: true,
        email: true,
        name: true,
        role: true,
        isActive: true,
        createdAt: true,
        merchantId: true,
      },
      orderBy: { createdAt: 'desc' },
    }),
    this.prisma.user.count({ where }),
  ]);

  return { users, total, page: dto.page || 1, limit: dto.limit || 20 };
}
```

**Verify:**
- Method filters by role
- Method filters by isActive status
- Method supports search by email/name
- Method supports pagination
- Method returns paginated results

### Task 3: Implement Suspend User

**Type:** feature
**Files:** `src/modules/admin/admin.service.ts`

**Action:**
Add suspendUser method to admin service:
```typescript
async suspendUser(dto: SuspendUserDto, adminId: string) {
  const user = await this.prisma.user.findUnique({
    where: { id: dto.userId },
  });

  if (!user) {
    throw new NotFoundException('User not found');
  }

  if (user.role === UserRole.ADMIN) {
    throw new ForbiddenException('Cannot suspend admin users');
  }

  if (!user.isActive) {
    throw new BadRequestException('User is already suspended');
  }

  const updated = await this.prisma.user.update({
    where: { id: dto.userId },
    data: {
      isActive: false,
      suspendedAt: new Date(),
      suspendedBy: adminId,
      suspensionReason: dto.reason,
    },
  });

  // Log audit entry
  await this.prisma.auditLog.create({
    data: {
      action: 'SUSPEND_USER',
      adminId,
      targetId: dto.userId,
      targetType: 'USER',
      details: { reason: dto.reason },
    },
  });

  return updated;
}
```

**Verify:**
- Method validates user exists
- Method prevents suspending admin users
- Method validates user is active
- Method updates user to inactive
- Method logs audit entry
- Method returns updated user

**Done:** Suspend user implemented

async activateUser(dto: ActivateUserDto, adminId: string) {
  const user = await this.prisma.user.findUnique({
    where: { id: dto.userId },
  });

  if (!user) {
    throw new NotFoundException('User not found');
  }

  if (user.isActive) {
    throw new BadRequestException('User is already active');
  }

  const updated = await this.prisma.user.update({
    where: { id: dto.userId },
    data: {
      isActive: true,
      suspendedAt: null,
      suspendedBy: null,
      suspensionReason: null,
      activatedBy: adminId,
      activatedAt: new Date(),
      activationNotes: dto.notes,
    },
  });

  // Log audit entry
  await this.prisma.auditLog.create({
    data: {
      action: 'ACTIVATE_USER',
      adminId,
      targetId: dto.userId,
      targetType: 'USER',
      details: { notes: dto.notes },
    },
  });

  return updated;
}
```

async getUserDetails(userId: string) {
  const user = await this.prisma.user.findUnique({
    where: { id: userId },
    select: {
      id: true,
      email: true,
      name: true,
      role: true,
      isActive: true,
      createdAt: true,
      merchantId: true,
      subscriptionActive: true,
      suspendedAt: true,
      suspendedBy: true,
      suspensionReason: true,
    },
  });

  if (!user) {
    throw new NotFoundException('User not found');
  }

  // Include related data based on role
  let relatedData = {};
  if (user.role === UserRole.MERCHANT_ADMIN || user.role === UserRole.MERCHANT_STAFF) {
    const merchant = await this.prisma.merchant.findUnique({
      where: { id: user.merchantId },
      select: {
        id: true,
        name: true,
        status: true,
      },
    });
    relatedData = { merchant };
  }

  return { ...user, ...relatedData };
}
```

**Verify:**
- Method returns user details
- Method includes suspension information
- Method includes merchant data for merchant users
- Method throws NotFoundException for non-existent users

  @Get('users')

    return this.adminService.listUsers(dto);
  }

  @Get('users/:id')
  async getUserDetails(@Param('id') id: string) {
    return this.adminService.getUserDetails(id);

    @Body() dto: SuspendUserDto,

  @Post('users/activate')

    @Body() dto: ActivateUserDto,

    return this.adminService.activateUser(dto, admin.id);
  }
}
