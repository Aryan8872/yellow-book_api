```typescript
async listFraudFlags(dto: ListFraudFlagsDto) {
  const where = dto.status ? { status: dto.status } : {};
  const skip = ((dto.page || 1) - 1) * (dto.limit || 20);

  const [fraudFlags, total] = await Promise.all([
    this.prisma.fraudFlag.findMany({
      where,
      skip,
      take: dto.limit || 20,
      include: {
        user: {
          select: {
            id: true,
            email: true,
            name: true,
          },
        },
        redemptionSession: {
          select: {
            id: true,
            code: true,
            offerId: true,
          },
        },
      },
      orderBy: { createdAt: 'desc' },
    }),
    this.prisma.fraudFlag.count({ where }),
  ]);

  return { fraudFlags, total, page: dto.page || 1, limit: dto.limit || 20 };
}
```

**Verify:**
- Method filters by status
- Method supports pagination
- Method includes related user and redemption session data
- Method returns paginated results

### Task 3: Implement Review Fraud Flag

**Type:** feature
**Files:** `src/modules/admin/admin.service.ts`

**Action:**
Add reviewFraudFlag method to admin service:
```typescript
async reviewFraudFlag(dto: ReviewFraudDto, adminId: string) {
  const fraudFlag = await this.prisma.fraudFlag.findUnique({
    where: { id: dto.fraudFlagId },
    include: {
      user: true,
      redemptionSession: true,
    },
  });

  if (!fraudFlag) {
    throw new NotFoundException('Fraud flag not found');
  }

  if (fraudFlag.status !== FraudFlagStatus.PENDING_REVIEW) {
    throw new BadRequestException('Fraud flag is not pending review');
  }

  const updated = await this.prisma.$transaction(async (tx) => {
    // Update fraud flag
    const updatedFlag = await tx.fraudFlag.update({
      where: { id: dto.fraudFlagId },
      data: {
        status: dto.decision === FraudReviewDecision.CONFIRMED 
          ? FraudFlagStatus.CONFIRMED 
          : FraudFlagStatus.DISMISSED,
        reviewedAt: new Date(),
        reviewedBy: adminId,
        reviewDecision: dto.decision,
        reviewNotes: dto.notes,
      },
    });

    // If confirmed, suspend the user
    if (dto.decision === FraudReviewDecision.CONFIRMED) {
      await tx.user.update({
        where: { id: fraudFlag.userId },
        data: {
          isActive: false,
          suspendedAt: new Date(),
          suspendedBy: adminId,
          suspensionReason: 'Fraud confirmed',
        },
      });
    }

    return updatedFlag;
  });

  // Log audit entry
  await this.prisma.auditLog.create({
    data: {
      action: 'REVIEW_FRAUD_FLAG',
      adminId,
      targetId: dto.fraudFlagId,
      targetType: 'FRAUD_FLAG',
      details: {
        decision: dto.decision,
        notes: dto.notes,
        userId: fraudFlag.userId,
      },
    },
  });

  return updated;
}
```

**Verify:**
- Method validates fraud flag exists
- Method validates fraud flag is pending review
- Method updates fraud flag status
- Method suspends user if fraud confirmed
- Method uses transaction for atomicity
- Method logs audit entry
- Method returns updated fraud flag

**Done:** Review fraud flag implemented

```typescript
import { ReviewFraudDto } from './dto/review-fraud.dto';

@Controller('admin')
@UseGuards(JwtAuthGuard, AdminOnlyGuard)
export class AdminController {
  // ... existing methods

  @Get('fraud-flags')
  async listFraudFlags(@Body() dto: ListFraudFlagsDto) {
    return this.adminService.listFraudFlags(dto);
  }

  @Get('fraud-flags/:id')
  async getFraudFlagDetails(@Param('id') id: string) {
    return this.adminService.getFraudFlagDetails(id);
  }

  @Post('fraud-flags/review')
  async reviewFraudFlag(
    @Body() dto: ReviewFraudDto,
    @CurrentUser() admin: AuthenticatedUser,
  ) {
    return this.adminService.reviewFraudFlag(dto, admin.id);
  }
}
```

**Verify:**
