### Task 2: Create Admin-Only Guard

**Type:** scaffold
**Files:** `src/modules/admin/guards/admin-only.guard.ts`

**Action:**
Create the guard for admin-only access:
```typescript
import { Injectable, CanActivate, ExecutionContext, ForbiddenException } from '@nestjs/common';
import { UserRole } from '@prisma/client';

@Injectable()
export class AdminOnlyGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    
    if (user.role !== UserRole.ADMIN) {
      throw new ForbiddenException('Admin access required');
    }
    
    return true;
  }
}
```

**Verify:**
- Guard implements CanActivate
- Guard checks for ADMIN role
- Guard throws ForbiddenException for non-admins

**Done:** Admin-only guard created

### Task 4: Create Admin Service

**Type:** scaffold
**Files:** `src/modules/admin/admin.admin.service.ts`

**Action:**
Create the admin service with basic methods:
```typescript
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../infrastructure/prisma/prisma.service';
import { MerchantStatus } from '@prisma/client';

@Injectable()
export class AdminService {
  constructor(private prisma: PrismaService) {}

  async getDashboardStats() {
    const [totalMerchants, pendingMerchants, totalUsers, activeOffers] = await Promise.all([
      this.prisma.merchant.count(),
      this.prisma.merchant.count({ where: { status: MerchantStatus.PENDING_REVIEW } }),
      this.prisma.user.count(),
      this.prisma.offer.count({ where: { isActive: true } }),
    ]);

    return {
      totalMerchants,
      pendingMerchants,
      totalUsers,
      activeOffers,
    };
  }

  async listMerchants(dto: ListMerchantsDto) {
    const where = dto.status ? { status: dto.status } : {};
    const skip = ((dto.page || 1) - 1) * (dto.limit || 20);

    const [merchants, total] = await Promise.all([
      this.prisma.merchant.findMany({
        where,
        skip,
        take: dto.limit || 20,
        include: {
          branches: true,
        },
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.merchant.count({ where }),
    ]);

    return { merchants, total, page: dto.page || 1, limit: dto.limit || 20 };
  }
}
```

**Verify:**
- Service created with basic methods
- Methods use PrismaService
- Methods return expected data structures

**Done:** Admin service created

### Task 5: Create Admin Controller

**Type:** scaffold
**Files:** `src/modules/admin/admin.controller.ts`

**Action:**
Create the admin controller with basic endpoints:
```typescript
import { Controller, Get, Post, Body, UseGuards } from '@nestjs/common';
import { AdminService } from './admin.service';
import { AdminOnlyGuard } from './guards/admin-only.guard';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { ListMerchantsDto } from './dto/list-merchants.dto';
import { ApproveMerchantDto } from './dto/approve-merchant.dto';
import { RejectMerchantDto } from './dto/reject-merchant.dto';

@Controller('admin')
@UseGuards(JwtAuthGuard, AdminOnlyGuard)
export class AdminController {
  constructor(private adminService: AdminService) {}

  @Get('dashboard')
  async getDashboard() {
    return this.adminService.getDashboardStats();
  }

  @Get('merchants')
  async listMerchants(@Body() dto: ListMerchantsDto) {
    return this.adminService.listMerchants(dto);
  }

  @Post('merchants/approve')
  async approveMerchant(@Body() dto: ApproveMerchantDto) {
    return this.adminService.approveMerchant(dto);
  }

  @Post('merchants/reject')
  async rejectMerchant(@Body() dto: RejectMerchantDto) {
    return this.adminService.rejectMerchant(dto);
  }
}
```

**Verify:**
- Controller created with basic endpoints
- Controller uses AdminOnlyGuard
- Controller uses JwtAuthGuard
- Endpoints use appropriate DTOs

**Done:** Admin controller created
