### Task 1: Implement Merchant Approval

**Type:** feature
**Files:** `src/modules/admin/admin.service.ts`

**Action:**
Add approveMerchant method to admin service:
```typescript
async approveMerchant(dto: ApproveMerchantDto, adminId: string) {
  const merchant = await this.prisma.merchant.findUnique({
    where: { id: dto.merchantId },
  });

  if (!merchant) {
    throw new NotFoundException('Merchant not found');
  }

  if (merchant.status !== MerchantStatus.PENDING_REVIEW) {
    throw new BadRequestException('Merchant is not pending review');
  }

  const updated = await this.prisma.merchant.update({
    where: { id: dto.merchantId },
    data: {
      status: MerchantStatus.APPROVED,
      approvedAt: new Date(),
      approvedBy: adminId,
      approvalNotes: dto.notes,
    },
  });

  // Log audit entry
  await this.prisma.auditLog.create({
    data: {
      action: 'APPROVE_MERCHANT',
      adminId,
      targetId: dto.merchantId,
      targetType: 'MERCHANT',
      details: { notes: dto.notes },
    },
  });

  return updated;
}
```

**Verify:**
- Method validates merchant exists
- Method validates merchant is PENDING_REVIEW
- Method updates merchant status to APPROVED
- Method logs audit entry
- Method returns updated merchant

**Done:** Merchant approval implemented

```typescript
      targetId: dto.merchantId,
      targetType: 'MERCHANT',
      details: { reason: dto.reason },
    },
  });

  return updated;
}
```

### Task 4: Update Admin Controller

**Type:** feature
**Files:** `src/modules/admin/admin.controller.ts`

**Action:**
Update controller to pass admin context to service:
```typescript
import { CurrentUser } from '../../common/decorators/current-user.decorator';
import type { AuthenticatedUser } from '../auth/auth.types';

@Controller('admin')
@UseGuards(JwtAuthGuard, AdminOnlyGuard)
export class AdminController {
  constructor(private adminService: AdminService) {}

  @Get('dashboard')
  async getDashboard() {
    return this.adminService.getDashboardStats();
  }

  @Get('merchants')
  async listMerchants(@Body() dto: ListMerchantsDto) {
    return this.adminService.listMerchants(dto);
  }

  @Post('merchants/approve')
  async approveMerchant(
    @Body() dto: ApproveMerchantDto,
    @CurrentUser() admin: AuthenticatedUser,
  ) {
    return this.adminService.approveMerchant(dto, admin.id);
  }

  @Post('merchants/reject')
  async rejectMerchant(
    @Body() dto: RejectMerchantDto,
    @CurrentUser() admin: AuthenticatedUser,
  ) {
    return this.adminService.rejectMerchant(dto, admin.id);
  }
}
```

**Verify:**
- Controller passes admin ID to service methods
- Controller uses @CurrentUser() decorator
- Controller methods updated for approval/rejection

**Done:** Admin controller updated
