```typescript
async listPayouts(dto: ListPayoutsDto) {
  const where: any = {};

  if (dto.status) {
    where.status = dto.status;
  }

  if (dto.merchantId) {
    where.merchantId = dto.merchantId;
  }

  const skip = ((dto.page || 1) - 1) * (dto.limit || 20);

  const [payouts, total] = await Promise.all([
    this.prisma.payout.findMany({
      where,
      skip,
      take: dto.limit || 20,
      include: {
        merchant: {
          select: {
            id: true,
            name: true,
          },
        },
      },
      orderBy: { createdAt: 'desc' },
    }),
    this.prisma.payout.count({ where }),
  ]);

  return { payouts, total, page: dto.page || 1, limit: dto.limit || 20 };
}
```

**Verify:**
- Method filters by status
- Method filters by merchant
- Method supports pagination
- Method includes merchant data

### Task 3: Implement Process Payout

**Type:** feature
**Files:** `src/modules/admin/admin.service.ts`

**Action:**
Add processPayout method to admin service:
```typescript
async processPayout(dto: ProcessPayoutDto, adminId: string) {
  const payout = await this.prisma.payout.findUnique({
    where: { id: dto.payoutId },
    include: {
      merchant: true,
    },
  });

  if (!payout) {
    throw new NotFoundException('Payout not found');
  }

  if (payout.status !== PayoutStatus.PENDING) {
    throw new BadRequestException('Payout is not pending');
  }

  const updated = await this.prisma.payout.update({
    where: { id: dto.payoutId },
    data: {
      status: PayoutStatus.PAID,
      paidAt: new Date(),
      paidBy: adminId,
      transactionId: dto.transactionId,
      paymentNotes: dto.notes,
    },
  });

  // Log audit entry
  await this.prisma.auditLog.create({
    data: {
      action: 'PROCESS_PAYOUT',
      adminId,
      targetId: dto.payoutId,
      targetType: 'PAYOUT',
      details: {
        merchantId: payout.merchantId,
        amount: payout.amount,
        transactionId: dto.transactionId,
      },
    },
  });

  return updated;
}
```

**Verify:**
- Method validates payout exists
- Method validates payout is pending
- Method updates payout status to PAID
- Method records payment details
- Method logs audit entry
- Method returns updated payout

```typescript
async generatePayoutReport(dto: GeneratePayoutReportDto) {
  const where: any = {};

  if (dto.startDate) {
    where.createdAt = { ...where.createdAt, gte: new Date(dto.startDate) };
  }

  if (dto.endDate) {
    where.createdAt = { ...where.createdAt, lte: new Date(dto.endDate) };
  }

  if (dto.merchantId) {
    where.merchantId = dto.merchantId;
  }

  const payouts = await this.prisma.payout.findMany({
    where,
    include: {
      merchant: {
        select: {
          id: true,
          name: true,
        },
      },
    },
    orderBy: { createdAt: 'desc' },
  });

  // Calculate totals
  const totalAmount = payouts.reduce((sum, p) => sum + p.amount, 0);
  const paidAmount = payouts.filter(p => p.status === PayoutStatus.PAID).reduce((sum, p) => sum + p.amount, 0);
  const pendingAmount = payouts.filter(p => p.status === PayoutStatus.PENDING).reduce((sum, p) => sum + p.amount, 0);

  return {
    payouts,
    summary: {
      totalPayouts: payouts.length,
      totalAmount,
      paidAmount,
      pendingAmount,
    },
  };
}
```

**Verify:**
- Method filters by date range
- Method filters by merchant
- Method returns all matching payouts
- Method calculates summary statistics
- Method returns summary with payouts

```typescript
import { ProcessPayoutDto } from './dto/process-payout.dto';

@Controller('admin')
@UseGuards(JwtAuthGuard, AdminOnlyGuard)
export class AdminController {
  // ... existing methods

  @Get('payouts')
  async listPayouts(@Body() dto: ListPayoutsDto) {
    return this.adminService.listPayouts(dto);
  }

  @Post('payouts/process')
  async processPayout(
    @Body() dto: ProcessPayoutDto,
    @CurrentUser() admin: AuthenticatedUser,
  ) {
    return this.adminService.processPayout(dto, admin.id);
  }

  @Post('payouts/report')
  async generatePayoutReport(@Body() dto: GeneratePayoutReportDto) {
    return this.adminService.generatePayoutReport(dto);
  }
}
```
