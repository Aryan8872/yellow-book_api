### Task 2: Create Haversine Distance Utility

**Type:** scaffold
**Files:** `src/modules/offer/offer.utils.ts`

**Action:**
Add Haversine distance calculation function to offer.utils.ts:
```typescript
/**
 * Calculate distance between two coordinates using Haversine formula
 * @returns Distance in kilometers
 */
export function haversineDistance(
  lat1: number,
  lon1: number,
  lat2: number,
  lon2: number
): number {
  const R = 6371; // Earth's radius in kilometers
  const dLat = toRad(lat2 - lat1);
  const dLon = toRad(lon2 - lon1);
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(toRad(lat1)) *
      Math.cos(toRad(lat2)) *
      Math.sin(dLon / 2) *
      Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

function toRad(degrees: number): number {
  return degrees * (Math.PI / 180);
}
```

**Verify:**
- Haversine function exists
- Returns distance in kilometers
- Uses Earth's radius of 6371km

**Done:** Haversine utility created

### Task 3: Add Location Filtering to Offer Service

**Type:** feature
**Files:** `src/modules/offer/offer.service.ts`

```typescript
import { haversineDistance } from './offer.utils';

async getOffers(
  query: QueryOffersDto,
  user?: AuthenticatedUser,
): Promise<{ offers: OfferSummary[]; count: number; total: number }> {
  const page = query.page ?? 1;
  const limit = Math.min(query.limit ?? 20, 100);
  const skip = (page - 1) * limit;

  const where: Prisma.OfferWhereInput = {
    isActive: true,
  };

  // Apply existing filters (category, search, etc.)
  // ... existing filter logic ...

  // Fetch offers with merchant branches
  const offers = await this.prisma.offer.findMany({
    where,
    include: {
      merchant: {
        include: {
          branches: true,
        },
      },
      category: true,
    },
    skip,
    take: limit,
    orderBy: { createdAt: 'desc' },
  });

  // Apply location filtering if coordinates provided
  let filteredOffers = offers;
  if (query.userLat !== undefined && query.userLng !== undefined && query.radius) {
    filteredOffers = offers.filter((offer) => {
      return offer.merchant.branches.some((branch) => {
        const distance = haversineDistance(
          query.userLat!,
          query.userLng!,
          Number(branch.lat),
          Number(branch.lng),
        );
        return distance <= query.radius!;
      });
    });
  }

  // Map to summary format
  const offerSummaries = filteredOffers.map((offer) =>
    mapOfferToSummary(offer),
  );

  const total = await this.prisma.offer.count({ where });
  const count = filteredOffers.length;

  return { offers: offerSummaries, count, total };
}
```

**Verify:**
- Location filtering applied when userLat, userLng, and radius provided
- Haversine distance calculated for each branch
- Offers filtered to those with at least one branch within radius
- Existing filters still work

**Done:** Location filtering added to offer service
